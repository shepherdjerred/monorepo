import DataTable from "../../../components/DataTable.astro";
import { scoutQlSourceCatalogs } from "@scout-for-lol/data/model/scoutql/catalog-columns.ts";

export const catalogs = scoutQlSourceCatalogs();

export const columnsOf = (catalog) => [...catalog.columns.values()];

export const validIn = (column) =>
  [
    column.contexts.select ? "SELECT" : null,
    column.contexts.where ? "WHERE" : null,
    column.contexts.groupBy ? "GROUP BY" : null,
  ]
    .filter((clause) => clause !== null)
    .join(" · ");

export const columnTable = (catalog) => [
  {
    header: "Column",
    values: columnsOf(catalog).map((column) => column.name),
    code: true,
  },
  {
    header: "Type",
    values: columnsOf(catalog).map((column) => column.type),
    code: true,
  },
  {
    header: "Origin",
    values: columnsOf(catalog).map((column) =>
      column.virtual ? "derived" : "match data",
    ),
  },
  { header: "Valid in", values: columnsOf(catalog).map(validIn) },
  {
    header: "Meaning",
    values: columnsOf(catalog).map((column) => column.description),
  },
];

`FROM` chooses the source. Its columns are the entire vocabulary available to
the rest of the query: what `SELECT` can aggregate, what `WHERE` can test, and
what `GROUP BY` can collapse on.

Not every column is valid everywhere, and the **Valid in** column below says
where each one may appear. A column that reads as a whole game — the queue, the
map, whether the game was surrendered — cannot always be a per-player
measurement, and a per-player measurement is not always a sensible grouping.

Columns marked **derived** are computed by Scout rather than stored: `player`,
`champion`, `patch`, and `outcome` exist because grouping by a raw id produces
a leaderboard of numbers.

`match_participants` is the right source for almost every report: one row per

### The time column

A source's time column is the one a [time
bound](/docs/reference/scoutql/#time-bounds-are-ordinary-where-conditions)
is measured on, and the one `DATE_TRUNC` buckets for a trend. Matches are
timestamped when the lobby was created (`game_creation_at`); champion-select
observations are timestamped when Scout saw them (`observed_at`).

The two rank sources have **no** time column, so they accept no window and no
time bucket. They are snapshots of where players stand right now — there is no
history in them to slice.

## Grouping

`GROUP BY` accepts any column marked valid there, plus expressions over them:

- `GROUP BY DATE_TRUNC('week', game_creation_at)` — a calendar bucket, which is
  what a line or area chart plots against.
- `GROUP BY FLOOR(game_duration_seconds / 300) * 300` — a numeric bucket, which
  is what a histogram counts into.
- `GROUP BY ALL` — DuckDB's shorthand for "group by every non-aggregate output".
- `GROUP BY group(2)` … `group(5)`, `group(all)` — only on `player_groups`,
  where the dimension is a set of tracked players who queued together.
- No `GROUP BY` at all — one grand-total row, which is what a KPI card shows.
